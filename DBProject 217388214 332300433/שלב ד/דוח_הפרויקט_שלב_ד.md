# דוח הפרויקט - שלב ד

בשלב זה התמקדנו בכתיבת תוכניות PL/pgSQL לביצוע פעולות לוגיות מורכבות על בסיס הנתונים שלנו, כולל פונקציות, פרוצדורות וטריגרים.

## שינויים במבנה בסיס הנתונים (Alter Table)
לצורך התוכניות השונות, בוצעו מספר שינויים בטבלאות הקיימות:
1. הוספת עמודת `Drone_Status` לטבלת הרחפנים כדי לנהל את זמינותם (`Available`, `In Mission`, `Maintenance`).
2. הוספת עמודת `Field_ID` לטבלת המשימות כדי לקשר באופן ישיר משימות הדברה לשדות הבעייתיים.

**הקוד (מתוך `AlterTable.sql`):**
```sql
-- Adding a status column to manage drone availability dynamically
ALTER TABLE DRONES 
ADD COLUMN Drone_Status VARCHAR(20) DEFAULT 'Available';

-- Adding a Field_ID to MISSIONS to directly connect local missions to remote agriculture fields
ALTER TABLE MISSIONS 
ADD COLUMN Field_ID INT;
```

## פונקציות

### 1. פונקציית `get_urgent_pest_reports`
**תיאור:** הפונקציה מקבלת רמת חומרה מינימלית ומחזירה מצביע (Ref Cursor) לרשומות של דיווחי מזיקים (מהטבלה המרוחקת `pest_reports`) שרמת החומרה שלהם שווה או גדולה לערך זה. נעשה כאן שימוש בהחזרת Ref Cursor.

**הקוד:**
```sql
CREATE OR REPLACE FUNCTION get_urgent_pest_reports(min_severity INT)
RETURNS refcursor AS $$
DECLARE
    report_cursor refcursor;
BEGIN
    OPEN report_cursor FOR
        SELECT report_id, field_id, pest_type, severity_level
        FROM pest_reports
        WHERE severity_level >= min_severity;
    RETURN report_cursor;
END;
$$ LANGUAGE plpgsql;
```

**הוכחה:**
הפונקציה מופעלת בהצלחה בתוך התוכנית הראשית הראשונה ("Emergency Dispatch"). ניתן לראות את הדפסת התוצאות מה-Cursor בצילום המסך של התוכנית הראשית למטה.

### 2. פונקציית `find_available_drone`
**תיאור:** הפונקציה מחפשת רחפן זמין שיכול לשאת משקל מסוים (המועבר כפרמטר). היא רצה בלולאה (Loop) על הרחפנים הזמינים, משתמשת בהסתעפות (IF) כדי לבדוק אם הרחפן מסוגל לשאת את המשקל, ואם כן היא עוצרת ומחזירה את מזהה הרחפן. במידה ולא נמצא רחפן מתאים - נזרקת חריגה (Exception).

**הקוד:**
```sql
CREATE OR REPLACE FUNCTION find_available_drone(required_payload_kg FLOAT)
RETURNS INT AS $$
DECLARE
    selected_drone_id INT := -1;
    drone_rec RECORD;
BEGIN
    FOR drone_rec IN 
        SELECT d.Drone_ID, dm.Max_Payload_Weight
        FROM DRONES d
        JOIN DRONE_MODELS dm ON d.Model_ID = dm.Model_ID
        WHERE d.Drone_Status = 'Available'
    LOOP
        IF drone_rec.Max_Payload_Weight >= required_payload_kg THEN
            selected_drone_id := drone_rec.Drone_ID;
            EXIT;
        END IF;
    END LOOP;

    IF selected_drone_id = -1 THEN
        RAISE EXCEPTION 'No available drone found capable of carrying % kg', required_payload_kg;
    END IF;

    RETURN selected_drone_id;
END;
$$ LANGUAGE plpgsql;
```

**הוכחה:**
הפונקציה מופעלת ונבדקת (כולל תפיסת החריגה במידה ואין רחפן) בתוך שתי התוכניות הראשיות (צילומי מסך למטה).

## פרוצדורות

### 1. פרוצדורת `auto_assign_urgent_missions`
**תיאור:** הפרוצדורה עוברת על דיווחי מזיקים חמורים בעזרת לולאת FOR (Implicit Cursor), מאתרת עבורם רחפן מתאים (ע"י הפעלת פונקציה מס' 2), ומייצרת משימת הדברה חדשה (INSERT). בנוסף היא מעדכנת את סטטוס הרחפן ל-'In Mission' (UPDATE). הפרוצדורה כוללת טיפול בשגיאות (Exception Handling) במקרה של כישלון בהקצאת רחפן כך שהלולאה ממשיכה לרוץ.

**הקוד:**
```sql
CREATE OR REPLACE PROCEDURE auto_assign_urgent_missions(min_severity INT)
LANGUAGE plpgsql AS $$
DECLARE
    report_rec RECORD;
    assigned_drone_id INT;
    new_mission_id INT;
    valid_client_id INT; 
BEGIN
    SELECT COALESCE(MAX(Mission_ID), 0) INTO new_mission_id FROM MISSIONS;
    SELECT COALESCE(MIN(Client_ID), 0) INTO valid_client_id FROM CLIENTS;

    FOR report_rec IN 
        SELECT report_id, field_id, severity_level 
        FROM pest_reports 
        WHERE severity_level >= min_severity
    LOOP
        BEGIN
            assigned_drone_id := find_available_drone(5.0);
            new_mission_id := new_mission_id + 1;
            
            INSERT INTO MISSIONS (Mission_ID, Mission_Title, Mission_Date, Mission_Type, Mission_Description, Mission_Status, Client_ID, Field_ID)
            VALUES (new_mission_id, 'Urgent Pest Control', CURRENT_DATE, 'Spraying', 'Auto-assigned mission for severe pest outbreak', 'Pending', valid_client_id, report_rec.field_id);
            
            UPDATE DRONES 
            SET Drone_Status = 'In Mission' 
            WHERE Drone_ID = assigned_drone_id;
            
            RAISE NOTICE 'Mission created successfully for report % with drone %', report_rec.report_id, assigned_drone_id;
            
        EXCEPTION 
            WHEN OTHERS THEN
                RAISE NOTICE 'Skipping report %: %', report_rec.report_id, SQLERRM;
        END;
    END LOOP;
END;
$$;
```

**הוכחה:**
ניתן לראות את ההדפסות והלוגים של יצירת המשימות בצילום המסך של התוכנית הראשית `main 1.png`.

### 2. פרוצדורת `flag_drones_for_maintenance`
**תיאור:** הפרוצדורה משתמשת בסמן מפורש (Explicit Cursor) כדי לעבור על כל הרחפנים שנרכשו לפני תאריך מסוים ועדיין מוגדרים כזמינים, ומשנה את הסטטוס שלהם ל-'Maintenance' (UPDATE).

**הקוד:**
```sql
CREATE OR REPLACE PROCEDURE flag_drones_for_maintenance(purchase_cutoff DATE)
LANGUAGE plpgsql AS $$
DECLARE
    drone_cursor CURSOR FOR 
        SELECT Drone_ID 
        FROM DRONES 
        WHERE Purchase_Date < purchase_cutoff AND Drone_Status = 'Available';
    current_drone_id INT;
BEGIN
    OPEN drone_cursor;
    LOOP
        FETCH drone_cursor INTO current_drone_id;
        EXIT WHEN NOT FOUND;
        
        UPDATE DRONES 
        SET Drone_Status = 'Maintenance' 
        WHERE Drone_ID = current_drone_id;
        
        RAISE NOTICE 'Drone ID % sent to maintenance.', current_drone_id;
    END LOOP;
    CLOSE drone_cursor;
END;
$$;
```

**הוכחה:**
ניתן לראות את הלוגים "Drone ID X sent to maintenance" בצילום המסך של התוכנית הראשית `main 2.png`.

## טריגרים

### 1. טריגר `trg_release_drone` (בזמן UPDATE)
**תיאור:** הטריגר פועל לאחר עדכון (AFTER UPDATE) בטבלת המשימות. במידה וסטטוס המשימה עודכן ל-'Completed', הוא יאתר את הרחפן שביצע את המשימה וישנה את הסטטוס שלו בחזרה ל-'Available'.

**הקוד:**
```sql
CREATE OR REPLACE FUNCTION release_drone_on_mission_complete()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.Mission_Status = 'Completed' AND OLD.Mission_Status <> 'Completed' THEN
        UPDATE DRONES
        SET Drone_Status = 'Available'
        WHERE Drone_ID IN (
            SELECT Drone_ID 
            FROM MISSION_ASSIGNMENTS 
            WHERE Mission_ID = NEW.Mission_ID
        );
        RAISE NOTICE 'Mission % completed. Drone released.', NEW.Mission_ID;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_release_drone
AFTER UPDATE ON MISSIONS
FOR EACH ROW
EXECUTE FUNCTION release_drone_on_mission_complete();
```

**הוכחה:**
*(לצורך הוכחת הטריגר הזה יש להוסיף כאן צילום מסך של שאילתת `UPDATE MISSIONS SET Mission_Status = 'Completed' WHERE Mission_ID = ...;` ולצידה שאילתת `SELECT` על טבלת הרחפנים שמראה שהסטטוס השתנה ל-Available)*

### 2. טריגר `trg_validate_purchase`
**תיאור:** טריגר הפועל לפני הוספת רחפן חדש (BEFORE INSERT) שמוודא שתאריך הרכישה לא יכול להיות בעתיד. במקרה שהוזן תאריך עתידי, נזרקת חריגה המונעת את ההכנסה לטבלה.

**הקוד:**
```sql
CREATE OR REPLACE FUNCTION validate_drone_purchase_date()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.Purchase_Date > CURRENT_DATE THEN
        RAISE EXCEPTION 'Invalid Purchase_Date: Cannot register a drone from the future (%).', NEW.Purchase_Date;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validate_purchase
BEFORE INSERT ON DRONES
FOR EACH ROW
EXECUTE FUNCTION validate_drone_purchase_date();
```

**הוכחה:**
*(לצורך הוכחה יש להוסיף צילום מסך של ניסיון הכנסת רחפן עם תאריך רכישה עתידי, לדוגמה `INSERT INTO DRONES (..., Purchase_Date) VALUES (..., '2030-01-01');` והודעת השגיאה שתקפוץ כתוצאה מכך)*

## תוכניות ראשיות

### 1. תוכנית ראשית 1: Emergency Dispatch
**תיאור:** תוכנית שקוראת לפונקציה הראשונה (לשליפת דיווחים דחופים) ומוכיחה את פעולת ה-Cursor שלה, ולאחר מכן מפעילה את הפרוצדורה הראשונה (לצורך הקצאת הרחפנים הזמינים ויצירת המשימות).

**הקוד:**
```sql
DO $$
DECLARE
    urgent_cursor refcursor;
    report_rec RECORD;
BEGIN
    RAISE NOTICE '--- Starting Main Program 1: Emergency Dispatch ---';
    urgent_cursor := get_urgent_pest_reports(4);
    FETCH urgent_cursor INTO report_rec;
    IF FOUND THEN
        RAISE NOTICE 'Function Success: Found urgent report for field % with severity %.', report_rec.field_id, report_rec.severity_level;
    ELSE
        RAISE NOTICE 'Function Success: No urgent reports found at the moment.';
    END IF;
    CLOSE urgent_cursor;
    
    RAISE NOTICE 'Calling Procedure 1 to auto-assign missions...';
    CALL auto_assign_urgent_missions(4);
    RAISE NOTICE '--- Main Program 1 Finished Successfully ---';
END;
$$;
```

### 2. תוכנית ראשית 2: Fleet Maintenance
**תיאור:** התוכנית מחפשת קודם כל רחפן פנוי להדגמת פעולת הפונקציה השנייה (ותפיסת חריגות אם אין), ולאחר מכן שולחת את הרחפנים הישנים לתחזוקה בעזרת הפרוצדורה השנייה.

**הקוד:**
```sql
DO $$
DECLARE
    available_drone_id INT;
BEGIN
    RAISE NOTICE '--- Starting Main Program 2: Fleet Maintenance ---';
    BEGIN
        available_drone_id := find_available_drone(3.5);
        RAISE NOTICE 'Function Success: Found available Drone ID % capable of carrying the payload.', available_drone_id;
    EXCEPTION 
        WHEN OTHERS THEN
            RAISE NOTICE 'Function Exception Caught: %', SQLERRM;
    END;
    
    RAISE NOTICE 'Calling Procedure 2 to flag old drones for maintenance...';
    CALL flag_drones_for_maintenance('2024-01-01');
    RAISE NOTICE '--- Main Program 2 Finished Successfully ---';
END;
$$;
```
